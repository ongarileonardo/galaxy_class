\section{Introduction}
\begin{frame}{Introduction}
    \begin{itemize}
    \setlength\itemsep{2em}
        \item The dataset used is the original Galaxy10 SDSS file 
          containing 21,785 RGB images of size 69×69 pixels and corresponding labels
        \item The purpose of the project is to use supervised learning techniques 
          to solve the classification task
        \item Different approaches will be explored:
          \begin{itemize}
            \item Galaxy10CNN model
            \item Raw Learning (no augmentation)
            \item Data augmentation
          \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Galaxy10CNN}
    \begin{itemize}
        \item Inspired by Jupyter Notebook intro provided by astroNN
        \item Built-in simple 4 layered convolutional neural network 
            with 2 conv. layers and 2 dense layers
    \begin{figure}
      \centering
      \includegraphics[width=0.5\textwidth]{../plots/galaxy10CNN.png}
    \end{figure}
    \end{itemize}
\end{frame}

\begin{frame}{User Defined CNN - Raw Data}
    \begin{itemize}
        \item 3 convolutional blocks with increasing depth (8, 32, 64 filters), 
            each followed by MaxPooling 
        \item Each block targets deeper features of the images
        \item This model works on ``raw'' data without rotation or flips
    \end{itemize}
    \begin{columns}
        \column{0.5\textwidth}

          \caption{Second layer feature maps}

    \end{columns}
\end{frame}

\begin{frame}{User Defined CNN - Data Augmentation}
    \begin{itemize}
    \setlength\itemsep{2em}
        \item Same format as the previous model, but with data augmentation applied
        \item Augmentation is done on the training set for each epoch and it includes:
        \begin{itemize}
            \item random flips (horizontally and vertically)
            \item random rotations (from -180 to 180 degrees)
        \end{itemize}
    \end{itemize}
\end{frame}

\section{Results}
\begin{frame}{Results}
    \begin{itemize}
    \setlength\itemsep{1em}
        \item All of the models were trained for 20 epochs
        \item Categorical cross-entropy loss and accuracy were used as metrics
        \item Galaxy10CNN was trained with tensorflow 2.18 for astroNN compatibility 
        \item The user-defined CNNs were trained with tensorflow 2.15 for GPU compatibility
        \item Gaussian smooth and augmentation were applied on user-defined CNNs only
    \end{itemize}
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion \& Future Work}
    \begin{itemize}
    \setlength\itemsep{1em}
        \item The user-defined CNNs showed slightly better performance than galaxy10CNN on the training set, 
            but same or worse on the validation set
        \item To prevent fluctuations, galaxy10CNN decreases the learning rate during training
        \item Since the data is unbalanced, the models do not capture every feature with the same weight
        \item Possible future work includes:
            \begin{itemize}
                \item Oversampling the minority classes
                \item Trying different losses and metrics
            \end{itemize}
    \end{itemize}
\end{frame}
